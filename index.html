import{state}from'./pdfViewer.js';export async function ocrMissingPages(onProgress){const words=[];for(const[n,items]of state.texts){if(items.some(i=>(i.str||'').trim()))continue;onProgress(`OCR page ${n}`);const c=state.pages.get(n).canvas;const r=await Tesseract.recognize(c,'eng',{logger:m=>m.status==='recognizing text'&&onProgress(`OCR page ${n}: ${Math.round(m.progress*100)}%`)});for(const w of r.data.words||[])words.push({page:n,text:w.text,bbox:w.bbox,canvas:c})}onProgress(words.length?'OCR ready':'OCR not required');return words}
